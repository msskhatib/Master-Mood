src,dst,img_how,img_vis=sys.argv[2:6]
d=docx.Document(src)
def allpars():
    for p in d.paragraphs: yield p
    seen=set()
    for t in d.tables:
        for row in t.rows:
            for c in row.cells:
                if c._tc in seen: continue
                seen.add(c._tc)
                for p in c.paragraphs: yield p
# 1. renumber figures (descending)
for old,new in [(8,10),(7,9),(6,8),(5,7),(4,6),(3,5),(2,3)]:
    for p in allpars():
        while re.search(r'Figure %d(?!\d)'%old,p.text):
            m=re.search(r'Figure %d(?!\d)'%old,p.text); replace_in_paragraph(p,m.group(0),'Figure@@%d'%new)
for p in allpars():
    for r in p.runs:
        if '@@' in r.text: r.text=r.text.replace('@@',' ')
def find(prefix):
    for p in d.paragraphs:
        if p.text.startswith(prefix): return p
    raise KeyError(prefix)
def set_text(pel, parts):
    """parts: list of (text,bold_or_None); reuse first run's rPr"""
    p=Paragraph(pel,None); runs=p.runs; base=runs[0]._r
    for r in runs[1:]: pel.remove(r._r)
    # remove non-run children like hyperlinks/bookmarks kept as is
    first=True
    for text,bold in parts:
        if first: r=base; first=False
        else:
            r=copy.deepcopy(base); pel.append(r)
        rr=docx.text.run.Run(r,p); rr.text=text
        if bold is not None: rr.bold=bold
cap1=find('Figure 1.'); imgp1=cap1._p.getprevious()
def new_image_par(path,width_in):
    el=copy.deepcopy(imgp1)
    rid,_=d.part.get_or_add_image(path)
    blip=el.find('.//'+A+'blip'); blip.set(R+'embed',rid)
    from PIL import Image
    w,h=Image.open(path).size; cx=int(Inches(width_in)); cy=int(cx*h/w)
    for e in el.iter(WP+'extent'): e.set('cx',str(cx)); e.set('cy',str(cy))
    for e in el.iter(A+'ext'):
        if e.get('cx'): e.set('cx',str(cx)); e.set('cy',str(cy))
    for e in el.iter(WP+'docPr'): e.set('id',str(900+int(width_in*100)%90)); 
    return el
