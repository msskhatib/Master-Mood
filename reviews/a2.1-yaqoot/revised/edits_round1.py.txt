def tfs(sl):
    for sh in sl.shapes:
        if sh.has_text_frame: yield sh.text_frame
        if sh.has_table:
            seen=set()
            for row in sh.table.rows:
                for c in row.cells:
                    if c._tc in seen: continue
                    seen.add(c._tc); yield c.text_frame
